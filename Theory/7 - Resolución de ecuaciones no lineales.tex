\section{Resolución de ecuaciones no lineales}

\subsection{Motivación}
Nuestro objetivo es partir de una función:
\[
    f: [a,b]\subset\mathbb{R}\to\mathbb{R}
\]
Continua, y encontrar sus raíces, es decir, los puntos $\xi_i \in [a,b]$:
\[
    f(\xi_i) = 0
\]

\subsection{Raíces aisladas}

\begin{definición}[Raíz aislada]
    Se dice que $\xi$ es una raíz aislada de $f$ si existe un entorno $U$ de $\xi$ tal que $\xi$ es la única raíz de $f$ en $U$, es decir, si $\exists \epsilon > 0$ tal que:
    \[
        f(x) \neq 0 \quad \forall x \in ( \xi - \epsilon, \xi + \epsilon ) \setminus \{ \xi \}
    \]   
\end{definición}

\begin{observación}
    Si una función es derivable, entonces todas sus raices son aisladas.
\end{observación}

\begin{teorema}[Algoritmo general]
    Veamos los pasos a seguir:
    \begin{enumerate}
        \item Aislamos las raíces, mediante intervalos encajados, de tal forma que:
        \[
            \forall i \exists ! \xi_i \in (a_i, b_i) 
        \]
        Esto se puede apreciar en la siguiente figura:
        \begin{center}
            \begin{tikzpicture}
                % Recta horizontal
                \draw[->] (-0.5, 0) -- (5.5, 0) node[right] {$x$};
                % puntos a y b
                \draw (0, 0) node[below] {$a$} -- (0, 0.5);
                \draw (5, 0) node[below] {$b$} -- (5, 0.5);

                % Raices xi1 y xi2
                \draw (1.5, 0) node[below] {$\xi_1$} -- (1.5, 0.5);
                \draw (3.5, 0) node[below] {$\xi_2$} -- (3.5, 0.5);

                % Intervalos (a1, b1) y (a2, b2)
                \draw (1, 0) node[below] {$a_1$} -- (1, 0.5);
                \draw (2, 0) node[below] {$b_1$} -- (2, 0.5);
                \draw (3, 0) node[below] {$a_2$} -- (3, 0.5);
                \draw (4, 0) node[below] {$b_2$} -- (4, 0.5);



            \end{tikzpicture}
        \end{center}
        \item Aproximar $\xi_i$ en cada intervalo $(a_i, b_i)$, mediante un método numérico iterativo, es decir:
        \[
            \begin{cases}
                x_0 \in (a_i, b_i) \\
                x_{n+1} = h(x_n) \quad \forall n \in \mathbb{N}
            \end{cases}
            \implies \lim_{n\to\infty} x_n = \xi_i
        \]
    \end{enumerate}
\end{teorema}

\subsection{Separación de raíces}

Recordemos algunos teoremas que nos servirán para separar raíces.

\begin{teorema}[Teorema de Bolzano]
    Si $f$ es continua en $[a,b]$ y $f(a)f(b)<0$, entonces existe $\xi \in (a,b)$ tal que $f(\xi)=0$.
\end{teorema}

\begin{teorema}[Teorema de Rolle]
    Si $f$ es continua en $[a,b]$, derivable en $(a,b)$ y $f(a)=f(b)$, entonces existe $\xi \in (a,b)$ tal que $f'(\xi)=0$.
\end{teorema}

\begin{teorema}[Teorema del valor medio]
    Si $f$ es continua en $[a,b]$ y derivable en $(a,b)$, entonces existe $\xi \in (a,b)$ tal que:
    \[
        f'(\xi) = \frac{f(b)-f(a)}{b-a}
    \]
\end{teorema}

Ahora veamos un teorema importante para separar y caracterizar raíces.

\begin{teorema}[Teorema de existencia y unicidad de raíces]
    Si $f$ es continua en $[a,b]$, derivable en $(a,b)$ y $f(a)f(b)<0$, y además $f'(x)$ tiene un signo constante en $(a,b)$, entonces:
    \[
        \begin{cases}
            f\in C^0([a,b]) \\
            f\in C^1((a,b)) \\
            f(a)f(b)<0 \\
            f'(x) \neq 0 \quad \forall x \in (a,b)
        \end{cases}
        \implies
        \exists ! \xi \in (a,b) \text{ tal que } f(\xi)=0
    \]   
\end{teorema}

Veamos como demostrarlo.

\begin{proof}
    Veamos la existencia y la unicidad por separado:
    \begin{itemize}
        \item \textbf{Existencia: } Como se cumplen las hipótesis del teorema de Bolzano, entonces existe $\xi \in (a,b)$ tal que $f(\xi)=0$.
        \item \textbf{Unicidad: } Supongamos que existen dos raíces distintas $\xi_1, \xi_2 \in (a,b)$, con $\xi_1 < \xi_2$. Entonces, por el teorema de Rolle, existe $\xi \in (\xi_1, \xi_2) \subset (a,b)$ tal que $f'(\xi)=0$, lo cual contradice la hipótesis de que $f'(x) \neq 0$ para todo $x \in (a,b)$. Por lo tanto, no pueden existir dos raíces distintas, es decir, la raíz es única.
    \end{itemize}
\end{proof}

\begin{teorema}[Teorema de separación de raíces]
    Si $f$ es continua en $[a,b]$, derivable en $(a,b)$ y $f(a)f(b)<0$, y además $f'(x)$ tiene $n$ raíces distintas en $(a,b)$, entonces $f$ tiene a lo sumo $n+1$ raíces en $(a,b)$, es decir:
    \[
        \begin{cases}
            f\in C^0([a,b]) \\
            f\in C^1((a,b)) \\
            f(a)f(b)<0 \\
            f'(x) \text{ tiene } n \text{ raíces distintas en } (a,b)
        \end{cases}
        \implies
        f \text{ tiene a lo sumo } n+1 \text{ raíces en } (a,b)
    \]
\end{teorema}

\begin{proof}
    Primero ordenemos las raíces de $f'$ de menor a mayor, es decir, $a < \xi_1 < \xi_2 < \ldots < \xi_n < b$. Entonces, si ahora tomamos los intervalos $[a, \xi_1], [\xi_1, \xi_2], \ldots, [\xi_n, b]$, en cada uno de estos intervalos, $f'$ no cambia de signo, y por tanto por el teorema de Rolle en cada intervalo, no podemos tener más de una raíz de $f$. Por lo tanto, en total, $f$ puede tener a lo sumo $n+1$ raíces en $(a,b)$.
\end{proof}

\begin{corolario}
    Si tenemos una función $f \in C^2([a,b])$ tal que $f''(x) > 0$ para todo $x \in (a,b)$, entonces $f$ tiene a lo sumo dos raíces en $(a,b)$.
\end{corolario}

\begin{proof}
    Si $f''(x) > 0$ para todo $x \in (a,b)$, entonces $f'$ es estrictamente creciente en $(a,b)$, lo que implica que $f'$ tiene a lo sumo una raíz en $(a,b)$. Por lo tanto, por el teorema de separación de raíces, $f$ tiene a lo sumo dos raíces en $(a,b)$.
\end{proof}

\subsection{Raíces de polinomios}

Veamos un teorema interesante sobre las raíces de los polinomios.

\begin{teorema}[Cota de las raíces de un polinomio]
    Si $P(x) = a_n x^n + a_{n-1} x^{n-1} + \ldots + a_1 x + a_0$ es un polinomio con $a_0 \neq 0 \neq a_n$, entonces todas sus raíces $\xi$ satisfacen:
    \[
        \begin{cases}
            \lambda = \max_{i \neq n } \left| \frac{a_i}{a_n} \right| \\
            \mu = \max_{i \neq 0 } \left| \frac{a_i}{a_0} \right|
        \end{cases}
        \implies
        \frac{1}{1+\mu} < |\xi| < 1 + \lambda
    \]

    Es decir, las raíces están fuera del disco de centro $0$ y radio $\frac{1}{1+\mu}$, y dentro del disco de centro $0$ y radio $1 + \lambda$.

    \begin{center}
        \begin{tikzpicture}
            % Ejes
            \draw[->] (-4, 0) -- (4, 0) node[right] {$\text{Re}$};
            \draw[->] (0, -4) -- (0, 4) node[above] {$\text{Im}$};

            % Discos
            \draw[thick, blue] (0, 0) circle (2.5);
            \draw[thick, red] (0, 0) circle (1.5);

            % Etiquetas
            \node at (2.5, 0.3) [above] {$1 + \lambda$};
            \node at (1.5, 0.3) [above] {$\frac{1}{1+\mu}$};
        \end{tikzpicture}
    \end{center}
    
\end{teorema}

\ejemplo{
    Sea el polinomio:
    \[
        p(x) = x^{100} - x^{50} +20x^{20} + 30
    \]
    Donde tenemos que:
    \[
        \begin{cases}
            a_n = 1 \implies \lambda = \max (\frac{1}{1}, \frac{20}{1}, \frac{30}{1}) = 30 \\
            a_0 = 30 \implies \mu = \max (\frac{1}{30}, \frac{1}{30}, \frac{20}{30}) = \frac{2}{3}
        \end{cases}
    \]
    Por lo tanto, las raíces de $p$ están fuera del disco de centro $0$ y radio $\frac{1}{1 + \frac{2}{3}} = \frac{3}{5}$, y dentro del disco de centro $0$ y radio $1 + 30 = 31$.
}

\begin{teorema}[Regla de Descartes]
    Sea $p_n \in \mathcal{P}_n$, con sus terminos siendo $\{ a_n \}_{i=0}^{n}$.\\
    Esta regla se basa en contar los cambios de signo, siendo esta cantidad $m$, y dejando que la cantidad de raices positivas es:
    \[
        \begin{cases}
            m \text{ es par: } \{m, m-2, \ldots 0\} \\
            m \text{ es impar: } \{m, m-2, \ldots 1\}
        \end{cases}
    \]
    Por ejemplo con el polinomio anterior que era:
    \[
        p(x) = x^{100} - x^{50} +20x^{20} + 30
    \]
    Tenemos la siguiente sucesión de terminos:
    \[
        \underbrace{30}_{+} \xrightarrow{0} \underbrace{20}_{+} \xrightarrow{+1}
        \underbrace{-1}_{-} \xrightarrow{+1} \underbrace{1}_{+}
    \]
    En este caso, ha habido 2 cambios en los signos de las raices, por lo que como es una cantidad par, habrá $2$ ó $0$ raices positivas del polinomio.\\
    Si ahora cambiamos al polinomio:
    \[
        p(x) = x^{100} - x^{50} +20x^{20} \textbf{-} 30
    \]
    Tenemos la siguiente sucesión de terminos:
    \[
        \underbrace{-30}_{-} \xrightarrow{+1} \underbrace{20}_{+} \xrightarrow{+1}
        \underbrace{-1}_{-} \xrightarrow{+1} \underbrace{1}_{+}
    \]
    Ahora habria 3 cambios de signos, por lo que puede haber $3$ ó $1$ raices positivas del polinomio.\\
    Para ver cuantas raices negativas hay, basta con evaluar el polinomio en $-x$, es decir:
    \[
        q(x) = p(-x) = (-1)^n a_n x^n + (-1)^{n-1} a_{n-1} x^{n-1} + \ldots + (-1)^0 a_0
    \]
    En nuestro caso, el polinomio se quedaría igual, es decir:
    \[
        q(x) = p(-x) =  x^{100} - x^{50} +20x^{20} - 30
    \]
    Por lo tanto, el número de cambios de signos en $q$ es el mismo que en $p$, por lo que también habría $3$ ó $1$ raices negativas del polinomio.
\end{teorema}

\subsection{Métodos numéricos}

Los métodos numéricos son iterativos.\\

En este caso ya hemos conseguido separar la única raiz $\xi$ de $f$ en un intervalo $[a,b]$, mediante la siguiente iteración:
\[
    \begin{cases}
        x_0 \in (a,b) \\
        x_{n+1} = h(x_n) \quad \forall n \in \mathbb{N}
    \end{cases}
    \implies
    \lim_{n\to\infty} x_n = \xi \iff \text{ el método converge}
\]

Siendo el error en la i-ésima iteración:
\[
    e_i = |x_i - \xi|
\]

\begin{definición}[Orden de un método numérico]
    Se existe una constante $M > 0$ y un número $\alpha > 0$ tal que:
    \[
        e_{i+1} \leq M e_i^\alpha \quad \forall i \in \mathbb{N}
    \]
    Entonces se dice que el método numérico tiene orden $\alpha$. En particular, distinguimos:
    \begin{itemize}
        \item \textbf{Orden 1: } Si $\alpha = 1$, entonces el método es de orden 1, y se dice que el método es linealmente.
        \item \textbf{Orden entre 1 y 2: } Si $1 < \alpha < 2$, entonces el método es de orden entre 1 y 2, y se dice que el método es superlineal.
        \item \textbf{Orden 2: } Si $\alpha = 2$, entonces el método es de orden 2, y se dice que el método es cuadrático.
    \end{itemize}
\end{definición}

\begin{teorema}[Método de la bisección]
    Si $f$ es continua en $[a,b]$, y $f(a)f(b)<0$, entonces el método de la bisección converge a la única raíz $\xi$ de $f$ en $(a,b)$ de la siguiente forma:
    \begin{enumerate}
        \item Sea $x_0 = \frac{a+b}{2}$.
        \item Calculamos $f(x_0)$, y si $f(x_0) = 0$, entonces hemos encontrado la raíz, es decir, $\xi = x_0$. Si no, proseguiremos al siguiente paso.
        \item[3.1] Si $f(a)f(x_0) < 0$, entonces la raíz $\xi$ se encuentra en el intervalo $(a, x_0)$, por lo que actualizamos $b = x_0$. Repetir.
        \item[3.2] Si $f(x_0)f(b) < 0$, entonces la raíz $\xi$ se encuentra en el intervalo $(x_0, b)$, por lo que actualizamos $a = x_0$. Repetir.
    \end{enumerate}
    Como cada vez dividimos el intervalo por la mitad, el error se puede calcular como:
    \[
        e_i = \frac{b-a}{2^{i+1}} \xrightarrow{i \to \infty} 0
    \]
    Y como:
    \[
        e_{i+1} = \frac{1}{2} e_1
    \]
    Dejando así que es un método lineal, al tener $\alpha = 1$ y $M=\frac{1}{2} < 1$.
\end{teorema}

\subsection{Métodos del punto fijo}

\begin{definición}[Aplicación contractiva]
    Se dice que una función $f: [a,b] \to \mathbb{R}$ es una aplicación contractiva si existe una constante $M \in [0,1)$ tal que:
    \[
        |f(x) - f(y)| \leq M |x-y| \quad \forall x,y \in [a,b]
    \]
    A la constante $M$ se le llama constante de contracción o constante de Lipschitz.
\end{definición}

\begin{corolario}[Continuidad de las aplicaciones contractivas]
    Si $f$ es una aplicación contractiva en $[a,b]$, entonces $f$ es continua en $[a,b]$.
\end{corolario}

Una pregunta natural sería como encontrar la constante de contracción $M$ de una aplicación contractiva $f$. Para ello, usaremos el teorema de Rolle. Definamos la siguiente función auxiliar:
\[
    \begin{array}{cccc}
        F: & [a,b] & \to & \mathbb{R} \\
        & z & \mapsto & F(z) = f(z) - \frac{f(x)(y-z) + f(y)(z-x)}{y-x}
    \end{array}
\]

Siendo $x,y \in [a,b]$ con $x < y$. Nótese que:
\[
    \begin{cases}
        F(x) = f(x) - \frac{f(x)(y-x) + f(y)(x-x)}{y-x} = 0 \\
        F(y) = f(y) - \frac{f(x)(y-y) + f(y)(y-x)}{y-x} = 0
    \end{cases}
\]
Esta función es claramente $C^1$ en $[a,b]$, por lo que por el teorema de Rolle, existe $c \in (x,y)$ tal que $F'(c) = 0$. Calculando la derivada de $F$, tenemos:
\[
    F'(c) = f'(c) - \frac{-f(x) + f(y)}{y-x} = 0 \implies
    f(y) - f(x) = f'(c)(y-x) \implies
    |f(y) - f(x)| = |f'(c)||y-x| \leq \| f'\|_\infty |y-x|
\]

Y así, finalmente, la constante $M$ será:
\[
    \boxed{M = \| f'\|_\infty}
\]

\ejemplo{
    Sea la función;
    \[
        f(x) = e^x - \sin(x)
    \]
    Con dominio $x \in [0, \frac{\pi}{2}]$. Calculemos entonces sus derivadas:
    \[
        \begin{cases}
            f'(x) = e^x - \cos(x) \\
            f''(x) = e^x + \sin(x)
        \end{cases}
    \]
    Como la 2 derivada es siempre positiva, entonces $f'$ es estrictamente creciente, y por tanto, el máximo de $f'$ en $[0, \frac{\pi}{2}]$ se encuentra en el extremo derecho del intervalo, es decir: 
    \[
        \| f' \|_\infty = f'(\frac{\pi}{2}) = e^{\frac{\pi}{2}}
    \]
    Que es una mejor aproximacion a simplemente acotar:
    \[
        \| f' \|_\infty = \| e^x - \cos(x) \|_\infty \leq \| e^x \|_\infty + \| \cos(x) \|_\infty = e^{\frac{\pi}{2}} + 1
    \]
}

\subsubsection{Teorema de Banach}

\begin{teorema}[Teorema de Banach]
    Sea $h$ una aplicación que cumple:
    \begin{enumerate}
        \item $h:[a,b] \to [a,b]$, es decir, deja el intervalo $[a,b]$ invariante.
        \item $h$ es contractiva con $M<1$.
    \end{enumerate}
    Entonces la aplicación:
    \[
        \begin{cases}
            x_0 \in [a,b] \\
            \vdots \\
            x_i = h(x_{i-1}) \\
            \xi = h(\xi)
        \end{cases}
    \]
    Tiene un único punto fijo $\xi \in [a,b]$, y toda sucesión $\{x_i\}_{i=0}^\infty$ con $x_0 \in [a,b]$ converge a $\xi$.\\
    Además, el error:
    \[
        E_i = |x_i - \xi| \leq \frac{M^i}{1-M}|x_1-x_0| \leq  \frac{M^i}{1-M}(b-a)
    \]
\end{teorema}

Una forma de poder aplicar el teorema de Banach para encontrar los 0 de una función $f$ es la siguiente:
\begin{enumerate}
    \item Sumamos: $x + f(x) = x$ y denotamos $h(x) = x + f(x)$. Ahora aplicamos el teorema de Banach a $h$, ya que buscamos su punto fijo $h(\xi) = \xi \implies x + f(x) = x \implies f(x) = 0$.
    \item Ahora intentamos aplicar el teorema de Banach, para ello, necesitamos un intervalo invariante $[a,b]$ y buscar $M= \|1+f'\|_\infty < 1$.
\end{enumerate}

En muchas situaciones, nos podemos encontrar con que $M$ es un poco mayor que 1, o que el intervalo no es invariante, pero podemos multiplicar por una constante $\alpha$ que haga que $M$ sea menor que 1, o que el intervalo sea invariante, es decir, podemos intentar aplicar el teorema de Banach a la función $h(x) = x + \alpha f(x)$, ya que las raíces no cambian de $f(x)$ a $\alpha f(x)$.

\subsubsection{Criterios de parada}

Ahora es interesante plantearnos cuales pueden ser los criterios de parada para este método. Para ello, fijamos una tolerancia $\epsilon \ll 1$ y despejamos:
\[
    E_n \leq \epsilon \implies
    \frac{M^n}{1-M}|x_1-x_0| \leq |\epsilon| \implies
    M^n \leq \frac{(1-M)\epsilon}{|x_1-x_0|} \implies
    n \log(M) \leq \log\left( \frac{(1-M)\epsilon}{|x_1-x_0|} \right) \implies
\]
\[
    \boxed{
        n = \left\lceil \frac{\log\left( \frac{(1-M)\epsilon}{|x_1-x_0|} \right)}{\log(M)} \right\rceil
    }
\]
