% Ejercicio 1
\ejercicio{resuelto}{profesor}{
    Hallar el polinomio $P$ que interpola la función $f(x) = \cos(\pi x)$ en los puntos $x_0 = 0$, $x_1 = \dfrac{1}{2}$, $x_2 = 1$, $x_3 = \dfrac{3}{2}$.
}{
    Si tenemos la función $f(x) = \cos(\pi x)$ y los puntos $x_0 = 0$, $x_1 = \dfrac{1}{2}$, $x_2 = 1$, $x_3 = \dfrac{3}{2}$, entonces construimos la tabla de valores:
    \begin{center}
        \begin{tabular}{|c|cccc|}
            \hline
            $x$ & $0$ & $\dfrac{1}{2}$ & $1$ & $\dfrac{3}{2}$ \\
            \hline
            $y$ & $1$ & $0$ & $-1$ & $0$ \\
            \hline
        \end{tabular}
    \end{center}
    Ahora para construir el polinomio de interpolación, podemos usar la fórmula de Lagrange:
    \[
        P(x) = \sum_{i=0}^3 y_i L_i(x)
    \]
    Donde:
    \[
        L_i(x) = \prod_{j=0, j \neq i}^3 \frac{x - x_j}{x_i - x_j}
    \]
    Procedamos a calcular cada $L_i(x)$:
    \begin{itemize}
        \item Para $i=0$:
        \[
            L_0(x) = \frac{(x - \frac{1}{2})(x - 1)(x - \frac{3}{2})}{(0 - \frac{1}{2})(0 - 1)(0 - \frac{3}{2})} = \frac{(x - \frac{1}{2})(x - 1)(x - \frac{3}{2})}{\frac{3}{4}} = \frac{4}{3}(x - \frac{1}{2})(x - 1)(x - \frac{3}{2})
        \]
        \item Para $i=1$ no es necesario ya que:
        \[
            y_1 = 0 \implies y_1 L_1(x) = 0
        \]
        \item Para $i=2$:
        \[
            L_2(x) = \frac{(x - 0)(x - \frac{1}{2})(x - \frac{3}{2})}{(1 - 0)(1 - \frac{1}{2})(1 - \frac{3}{2})} = \frac{x(x - \frac{1}{2})(x - \frac{3}{2})}{-\frac{1}{4}} = -4x(x - \frac{1}{2})(x - \frac{3}{2})
        \]
        \item Para $i=3$ no es necesario ya que:
        \[
            y_3 = 0 \implies y_3 L_3(x) = 0
        \]
    \end{itemize}
    Dejando el polinomio:
    \[
        P(x) = 1 \cdot L_0(x) + (-1) \cdot L_2(x) = \frac{4}{3}(x - \frac{1}{2})(x - 1)(x - \frac{3}{2}) + 4x(x - \frac{1}{2})(x - \frac{3}{2})
    \]
}

% Ejercicio 3
\ejercicio{resuelto}{profesor}{
    Sean $\{x_0, x_1, \ldots, x_n\} \subset \mathbb{R}$ puntos distintos. Demostrar:
    \begin{enumerate}[label=(\alph*)]
        \item
        \[
            \sum_{i=0}^{n} L_i(x) = 1, \quad x \in \mathbb{R}.
        \]
        \item
        \[
            E_n(x) = f(x) - P_n(x) = \sum_{i=0}^{n} \bigl(f(x) - f(x_i)\bigr)L_i(x).
        \]
    \end{enumerate}
}{
    Recordemos que los polinomios básicos de Lagrange están dados por
    \[
        L_i(x)=\prod_{\substack{j=0 \\ j\neq i}}^{n}\frac{x-x_j}{x_i-x_j},\qquad i=0,\dots,n,
    \]
    y satisfacen la propiedad
    \[
        L_i(x_k)=\delta_{ik}=\begin{cases}
            1, & i=k,\\
            0, & i\neq k.
        \end{cases}
    \]

    \begin{enumerate}[label=(\alph*)]
        \item Consideremos
        \[
            S(x)=\sum_{i=0}^{n}L_i(x).
        \]
        Como cada $L_i$ tiene grado a lo sumo $n$, también $S$ tiene grado a lo sumo $n$.

        Evaluamos en los nodos $x_k$ ($k=0,\dots,n$):
        \[
            S(x_k)=\sum_{i=0}^{n}L_i(x_k)=\sum_{i=0}^{n}\delta_{ik}=1.
        \]
        Es decir, $S$ y el polinomio constante $1$ (de grado $\le n$) coinciden en los $n+1$ puntos distintos $x_0,\dots,x_n$.

        Por unicidad del polinomio interpolador de grado $\le n$, se concluye que
        \[
            \sum_{i=0}^{n}L_i(x)=S(x)=1,\qquad \forall x\in\mathbb{R}.
        \]

        \item Partimos de la expresión de Lagrange del interpolador:
        \[
            P_n(x)=\sum_{i=0}^{n}f(x_i)L_i(x).
        \]
        Entonces
        \[
            E_n(x)=f(x)-P_n(x)=f(x)-\sum_{i=0}^{n}f(x_i)L_i(x).
        \]
        Usando el apartado anterior, $\sum_{i=0}^{n}L_i(x)=1$, podemos escribir
        \[
            f(x)=f(x)\sum_{i=0}^{n}L_i(x)=\sum_{i=0}^{n}f(x)L_i(x).
        \]
        Sustituyendo en la expresión del error:
        \[
            E_n(x)=\sum_{i=0}^{n}f(x)L_i(x)-\sum_{i=0}^{n}f(x_i)L_i(x)
            =\sum_{i=0}^{n}\bigl(f(x)-f(x_i)\bigr)L_i(x).
        \]
        Por tanto,
        \[
            E_n(x)=f(x)-P_n(x)=\sum_{i=0}^{n}\bigl(f(x)-f(x_i)\bigr)L_i(x).
        \]
    \end{enumerate}
}

% Ejercicio 4
\ejercicio{resuelto}{profesor}{
    Sean $P\in\mathbb{P}_n$ y $n+1$ puntos distintos $\{x_0,x_1,\ldots,x_n\}$. Hallar el valor de
    \[
        P[x_0,x_1,\ldots,x_n].
    \]
}{
    Tengamos por ejemplo:
    \[
        p_1 = 5 + 2x \, \quad \{ x_0 = 3, x_1 = 9 \}
    \]
    Podemos calcular la diferencia dividida de $p_1$ en los puntos $x_0$ y $x_1$:
    \[
        p_1[x_0, x_1] = \frac{p_1(x_1) - p_1(x_0)}{x_1 - x_0} = \frac{(5 + 2 \cdot 9) - (5 + 2 \cdot 3)}{9 - 3} = \frac{23 - 11}{6} = 2.
    \]
    En general, si tenemos:
    \[
        p_1 = a_0 + a_1 x \, \quad \{ x_0, x_1 \}
    \]
    Podemos calcular la diferencia dividida de $p_1$ en los puntos $x_0$ y $x_1$:
    \[
        p_1[x_0, x_1] = \frac{p_1(x_1) - p_1(x_0)}{x_1 - x_0} = \frac{(a_0 + a_1 x_1) - (a_0 + a_1 x_0)}{x_1 - x_0} = \frac{a_1 (x_1 - x_0)}{x_1 - x_0} = a_1.
    \]
    Ahora si tomamos $f = p_n$, tenemos que:
    \[
        p_n(x) = p_{n-1}(x) + \underbrace{\underbrace{p_n[x_0, x_1, \ldots, x_n]}_{\alpha} \prod_{j=0}^{n-1} (x - x_j)}_{q_n(x)}.
    \]  

}

% Ejercicio 5
\ejercicio{resuelto}{profesor}{
    Sean $\{x_0,x_1,\ldots,x_n\}$ con $x_i\neq x_j$ si $i\neq j$ y $f(x)=x^{n+1}$. Calcular el polinomio de
    interpolaci\'on de Lagrange de $f$ en los puntos $\{x_0,x_1,\ldots,x_n\}$ y determinar su t\'ermino independiente.
}{
    Sabemos que el polinomio de interpolación de Lagrange de $f(x) = x^{n+1}$ en los puntos $\{x_0, x_1, \ldots, x_n\}$ es:
    \[
        P_n(x) = 
        \sum_{i=0}^{n} f(x_i) L_i(x) = 
        \sum_{i=0}^{n} x_i^{n+1} \cdot \prod_{\substack{j=0 \\ j \neq i}}^{n} \frac{x - x_j}{x_i - x_j}.
    \]
    Ahora podemos calcular el error de interpolación, que es:
    \[
        E_n(x) = 
        f(x) - P_n(x) = 
        \frac{(x^{n+1})^{(n+1)}}{(n+1)!} \prod_{i=0}^{n} (x - x_i) = 
        \frac{(n+1)!}{(n+1)!} \prod_{i=0}^{n} (x - x_i) = 
        \prod_{i=0}^{n} (x - x_i).
    \]
    Despejando nos queda:
    \[
        P_n(x) = f(x) - E_n(x) = x^{n+1} - \prod_{i=0}^{n} (x - x_i).
    \]
    Para determinar el término independiente de $P_n$, evaluamos en $x=0$:
    \[
        P_n(0) = 0^{n+1} - \prod_{i=0}^{n} (0 - x_i) = -(-1)^{n+1} \prod_{i=0}^{n} x_i = (-1)^n \prod_{i=0}^{n} x_i.
    \]
    Y por tanto, el término independiente de $P_n$ es $(-1)^n \prod_{i=0}^{n} x_i$.
}

% Ejercicio 8
\ejercicio{resuelto}{profesor}{
    Determinar los valores de $\lambda$ y $\mu$ para que
    \[
        S(x)=
        \begin{cases}
            \lambda x(x^2+1), & 0\le x\le 1,\\
            -\lambda x^3+\mu x^2-5\lambda x+1, & 1\le x\le 2,
        \end{cases}
    \]
    sea una funci\'on spline c\'ubica.
}{
    Para que $S$ sea una función spline cúbica, debe ser de clase $C^2$ en el intervalo $[0, 2]$. Esto implica que $S$ debe ser continua, y sus primeras y segundas derivadas también deben ser continuas en el punto de unión $x=1$ (en el resto del intervalo, cada parte es un polinomio cúbico, por lo que no hay problemas de continuidad allí).
    \begin{itemize}
        \item \textbf{Continuidad de $S$ en $x=1$:}
        \[
            \begin{cases}
                S(x=1-0) = \lambda \cdot 1 \cdot (1^2 + 1) = 2\lambda,\\
                S(x=1+0) = -\lambda \cdot 1^3 + \mu \cdot 1^2 - 5\lambda \cdot 1 + 1 = -6\lambda + \mu + 1.
            \end{cases}
            \implies 2\lambda = -6\lambda + \mu + 1 \implies \mu = 8\lambda - 1.
        \]
        \item \textbf{Continuidad de $S'$ en $x=1$:}
        Calculamos $S'$ para cada parte:
        \[
            S'(x) =
            \begin{cases}
                \lambda (3x^2 + 1), & 0 \le x \le 1,\\
                -3\lambda x^2 + 2\mu x - 5\lambda, & 1 \le x \le 2.
            \end{cases}
            \implies
        \]
        \[
            \begin{cases}
                S'(x=1-0) = \lambda \cdot (3 \cdot 1^2 + 1) = 4\lambda,\\
                S'(x=1+0) = -3\lambda \cdot 1^2 + 2\mu \cdot 1 - 5\lambda = -8\lambda + 2\mu.
            \end{cases}
            \implies
        \]
        \[
            4\lambda = -8\lambda + 2\mu \implies 12\lambda = 2\mu \implies \mu = 6\lambda.
        \]
        \item \textbf{Resolvemos el sistema de ecuaciones:}
        \[
            \begin{cases}
                \mu = 8\lambda - 1,\\
                \mu = 6\lambda.
            \end{cases}
            \implies 
            \begin{cases}
                6\lambda = 8\lambda - 1 \implies -2\lambda = -1 \implies \boxed{\lambda = \frac{1}{2}},\\
                \mu = 6\lambda = 6 \cdot \frac{1}{2} = 3 \implies \boxed{\mu = 3}.
            \end{cases}
        \]
        Por lo tanto, los valores de $\lambda$ y $\mu$ para que $S$ sea una función spline cúbica son $\lambda = \frac{1}{2}$ y $\mu = 3$, pero falta comprobar la continuidad de $S''$ en $x=1$.
        \item \textbf{Continuidad de $S''$ en $x=1$:}
        Calculamos $S''$ para cada parte:
        \[
            S''(x) =
            \begin{cases}
                \lambda (6x), & 0 \le x \le 1,\\
                -6\lambda x + 2\mu, & 1 \le x \le 2.
            \end{cases}
            \implies
        \]
        \[
            \begin{cases}
                S''(x=1-0) = \lambda \cdot 6 \cdot 1 = 6\lambda,\\
                S''(x=1+0) = -6\lambda \cdot 1 + 2\mu = -6\lambda + 2\mu.
            \end{cases}
            \implies
        \]
        \[
            6\lambda = -6\lambda + 2\mu \implies 
            12\lambda = 2\mu \implies 
            \mu = 6\lambda \impliedby 
            \begin{cases}
                \lambda = \frac{1}{2},\\
                \mu = 3.
            \end{cases}
        \]
        Este resultado coincide con el obtenido anteriormente, por lo que no hay contradicciones y los valores de $\lambda$ y $\mu$ son correctos.
        \item \textbf{Condiciones de contorno:}
        No se han especificado condiciones de contorno adicionales, por lo que no es necesario verificar nada más. Sin embargo, si se hubieran dado condiciones de contorno, habría que asegurarse de que $S$ las cumpla. En particular, si $S$ es de \textbf{Tipo I} entonces se deberían cumplir $S(0) = S(2) = 0$, lo cual no es el caso aquí, por lo que $S$ no es de \textbf{Tipo I}.
    \end{itemize}
}
